    get_target_property(sr_definitions ${sr_target} ${sr_property})
    if(sr_definitions)
      list(REMOVE_ITEM sr_definitions SFIZZ_ENABLE_RELEASE_DBG=1 SFIZZ_ENABLE_RELEASE_ASSERT=1)
      set_property(TARGET ${sr_target} PROPERTY ${sr_property} "${sr_definitions}")
    endif()
  endforeach()
